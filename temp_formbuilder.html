Sample form This example demonstrates a dynamic form with dependencies using the GC Design System components.

Provide a summary (you can format text, add lists, etc.)

 Submit Form 
var form = new FormDefinition
{
    Id = "demo-form",
    Title = "Dynamic Form Demo",
    Action = Url.Action("FormBuilder", "Components") ?? "",
    Method = "post",
    SubmitButtonText = "Submit Form",
    Sections = new List
    {
        new FormSection
        {
            Title = "Personal Information",
            Hint = "Please provide your basic information",
            Questions = new List
            {
                new FormQuestion
                {
                    Id = "fullName",
                    Label = "Full Name",
                    Type = QuestionType.Text,
                    IsRequired = true,
                    Hint = "Enter your full legal name"
                },
                (…)
            }
        }
    }
}
